คลิกแท็บ Form1.cs แล้วป้อนโปรแกรมต่อไปนี้ลงในหน้าจอ

using CF9000ControlModule;

เพิ่มโพรเซสต่อไปนี้ลงในโพรเซสที่เริ่มทำงานเมื่อกดปุ่ม

private void button1_Click (object sender, EventArgs e)
{
  // IP Address of FFT Analyzer
  cfController.SetIPAddress ("192.168.3.59");
  // Start
  cfController.Execute ("KeySTART");
}
